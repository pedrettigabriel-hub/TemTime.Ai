================================================================== # HTTP E SCRAPING # ================================================================== def http_get(session, url, params): for attempt in range(1, MAX_RETRIES + 1): try: r = session.get(url, params=params, headers=HEADERS, timeout=TIMEOUT) r.raise_for_status() ct = r.headers.get("Content-Type", "") if "application/json" not (requests.RequestException, json.JSONDecodeError, ValueError) as e: if attempt == MAX_RETRIES: raise RuntimeError(f"Falha definitiva: {url}") from e wait = min(2 ** attempt, 30) log.warning("Falha fim.", page) break novos = 0 for r in regs: rid = r.get("id") if rid is not None and rid not in vistos: vistos.add(rid) todos.append(r) novos += 1 log.info("Página %d: %d itens (%d novos, acumulado %d)", page, len(regs), novos, len(todos)) if len(regs) < PAGE_SIZE or novos == 0: break page += 1 time.sleep(SLEEP_BETWEEN) return todos # ================================================================== # CSV / JSON # ================================================================== def achatar(obj, prefixo=""): saida = {} if isinstance(obj, dict): for k, v in obj.items(): chave = f"{prefixo}.{k}" if prefixo else k if isinstance(v, dict): saida.update(achatar(v, prefixo=chave)) elif isinstance(v, list): saida[chave] = json.dumps(v, ensure_ascii=False) else: saida[chave] = v else: saida[prefixo or "valor"] = obj return saida def salvar_json(dados, caminho): caminho.write_text(json.dumps(dados, ensure_ascii=False,
